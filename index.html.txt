        let currentCurrency = 'USD';
        let selectedGame = null;
        let selectedPackage = null;
        let cart = [];
        let selectedWalletInfo = {
            name: 'Binance Pay',
            address: '0x71C7656EC7ab88b098defB751B7401B5f6d8976F'
        };

        function renderGames(data) {
            const container = document.getElementById('games-container');
            container.innerHTML = '';
            
            data.forEach(game => {
                const card = document.createElement('div');
                card.className = 'game-card';
                card.innerHTML = `
                    <div class="game-img-wrapper">
                        <img src="${game.image}" class="game-img" alt="${game.name}">
                    </div>
                    <div class="game-info">
                        <div class="game-title">${game.name}</div>
                        <button class="charge-btn" onclick="openPackageModal('${game.id}')">شحن الآن</button>
                    </div>
                `;
                container.appendChild(card);
            });
        }

        function setCurrency(curr) {
            currentCurrency = curr;
            document.getElementById('btn-usd').classList.toggle('active', curr === 'USD');
            document.getElementById('btn-lyd').classList.toggle('active', curr === 'LYD');
            updateCartUI();
        }

        function filterGames() {
            const query = document.getElementById('search-input').value.toLowerCase();
            const filtered = gamesData.filter(g => g.name.toLowerCase().includes(query));
            renderGames(filtered);
        }

        function openPackageModal(gameId) {
            selectedGame = gamesData.find(g => g.id === gameId);
            selectedPackage = null;
            
            document.getElementById('modal-game-title').innerText = selectedGame.name;
            const container = document.getElementById('packages-container');
            container.innerHTML = '';

            selectedGame.packages.forEach(pkg => {
                const price = currentCurrency === 'USD' ? `$${pkg.priceUSD}` : `${pkg.priceLYD} د.ل`;
                const item = document.createElement('div');
                item.className = 'pkg-item';
                item.innerHTML = `
                    <div class="pkg-title">${pkg.name}</div>
                    <div class="pkg-price">${price}</div>
                `;
                item.onclick = () => {
                    document.querySelectorAll('.pkg-item').forEach(el => el.classList.remove('selected'));
                    item.classList.add('selected');
                    selectedPackage = pkg;
                };
                container.appendChild(item);
            });

            document.getElementById('package-modal').style.display = 'flex';
        }

        function closeModal(modalId) {
            document.getElementById(modalId).style.display = 'none';
        }

        function addToCart() {
            const playerId = document.getElementById('player-id').value;
            if (!playerId) {
                alert('يرجى إدخال رقم اللاعب (Player ID)');
                return;
            }
            if (!selectedPackage) {
                alert('يرجى اختيار باقة لشحنها');
                return;
            }

            cart.push({
                gameName: selectedGame.name,
                package: selectedPackage,
                playerId: playerId
            });

            closeModal('package-modal');
            updateCartUI();
            toggleCart(true);
        }

        function toggleCart(forceOpen = false) {
            const drawer = document.getElementById('cart-drawer');
            if (forceOpen) {
                drawer.classList.add('open');
            } else {
                drawer.classList.toggle('open');
            }
        }

        function removeFromCart(index) {
            cart.splice(index, 1);
            updateCartUI();
        }

        function updateCartUI() {
            document.getElementById('cart-count').innerText = cart.length;
            const container = document.getElementById('cart-items-container');
            container.innerHTML = '';

            let total = 0;

            cart.forEach((item, index) => {
                const price = currentCurrency === 'USD' ? item.package.priceUSD : item.package.priceLYD;
                total += price;

                const el = document.createElement('div');
                el.className = 'cart-item';
                el.innerHTML = `
                    <div>
                        <div style="font-weight:bold;">${item.gameName}</div>
                        <div style="font-size:0.85rem; color:var(--text-muted);">${item.package.name} (ID: ${item.playerId})</div>
                        <div style="color:var(--accent-cyan); font-weight:bold;">${currentCurrency === 'USD' ? '$'+price : price+' د.ل'}</div>
                    </div>
                    <button onclick="removeFromCart(${index})" style="background:none; border:none; color:var(--accent-pink); cursor:pointer;"><i class="fa-solid fa-trash"></i></button>
                `;
                container.appendChild(el);
            });

            document.getElementById('cart-total-price').innerText = currentCurrency === 'USD' ? `$${total.toFixed(2)}` : `${total.toFixed(2)} د.ل`;
        }

        function selectWallet(name, address, element) {
            document.querySelectorAll('.wallet-btn').forEach(btn => btn.classList.remove('active'));
            element.classList.add('active');
            selectedWalletInfo = { name, address };
        }

        function openPaymentModal() {
            if (cart.length === 0) {
                alert('السلة فارغة!');
                return;
            }
            document.getElementById('pay-wallet-name').innerText = selectedWalletInfo.name;
            document.getElementById('wallet-address').innerText = selectedWalletInfo.address;
            document.getElementById('payment-modal').style.display = 'flex';
        }

        function confirmPayment() {
            const txid = document.getElementById('txid-input').value;
            if (!txid) {
                alert('يرجى إدخال رقم العملية / إثبات التحويل');
                return;
            }

            alert('تم استلام طلب الشحن بنجاح! سيتم التحقق من التحويل وإرسال الشدات/الجواهر فوراً.');
            cart = [];
            updateCartUI();
            closeModal('payment-modal');
            toggleCart();
        }

        renderGames(gamesData);
